/*
 * Leaves-WebUI nav rail.
 *
 * Loaded by any mod page that opts in with <script defer src="/webui/rail.js"></script>. Paints a
 * fixed vertical rail down the RIGHT edge listing every installed Leaves mod that has a web UI, with
 * the current one highlighted; clicking navigates to that mod's own page.
 *
 * Self-contained: no framework, no external fetch (only same-origin probes to localhost mods), its own
 * scoped `lwr-` styles so it can't collide with the host page. If a mod isn't installed its probe
 * fails and it's silently dropped, so the rail only ever shows what's actually there.
 */
(function () {
  "use strict";

  // ---- shared boolean search helper -----------------------------------------------------
  // Every mod page filters some list by a text box. This exposes a tiny query language all of
  // them share: `&` = AND, `|` = OR, with `&` binding tighter than `|`, so "a & b | c" means
  // "(a AND b) OR c". Plain text with no operators is just a single substring term, exactly as
  // before. Terms are matched case-insensitively against a page-supplied haystack string (each
  // page joins whatever fields it wants searchable into one string). Published on `window` so
  // every page — this rail loads on all of them — can call it without duplicating the parser.
  var LeavesSearch = {
    // "a & b | c" -> [["a","b"],["c"]]. Empty/whitespace terms are dropped; a fully empty query
    // returns [] which every caller treats as "match everything".
    parse: function (raw) {
      var groups = [];
      var text = (raw == null ? "" : String(raw)).toLowerCase();
      var ors = text.split("|");
      for (var i = 0; i < ors.length; i++) {
        var terms = [];
        var ands = ors[i].split("&");
        for (var j = 0; j < ands.length; j++) {
          var term = ands[j].trim();
          if (term) {
            terms.push(term);
          }
        }
        if (terms.length > 0) {
          groups.push(terms);
        }
      }
      return groups;
    },

    // Every term across all groups, flattened - for highlighting each matched piece.
    terms: function (raw) {
      var groups = this.parse(raw);
      var out = [];
      for (var i = 0; i < groups.length; i++) {
        for (var j = 0; j < groups[i].length; j++) {
          out.push(groups[i][j]);
        }
      }
      return out;
    },

    // Does `haystack` satisfy the query? True for an empty query. A group (AND) passes only if
    // every one of its terms is a substring; the whole query (OR) passes if any group passes.
    test: function (haystack, raw) {
      var groups = this.parse(raw);
      if (groups.length === 0) {
        return true;
      }
      var hay = (haystack == null ? "" : String(haystack)).toLowerCase();
      for (var i = 0; i < groups.length; i++) {
        var all = true;
        for (var j = 0; j < groups[i].length; j++) {
          if (hay.indexOf(groups[i][j]) === -1) {
            all = false;
            break;
          }
        }
        if (all) {
          return true;
        }
      }
      return false;
    }
  };
  window.LeavesSearch = LeavesSearch;

  // The known suite. Adding a mod later = add a row here (and drop the one <script> tag in its page).
  // `slug` is the mod's URL prefix, used both to probe and to highlight the current page.
  var MODS = [
    { slug: "questrandomizer", name: "Quest Randomizer", icon: "🎲", pageUrl: "/questrandomizer/ui" },
    { slug: "bartereconomy", name: "Barter Economy", icon: "🤝", pageUrl: "/bartereconomy/ui" },
    { slug: "lootfuckery", name: "Loot Fuckery", icon: "📦", pageUrl: "/lootfuckery/ui/dump" },
    { slug: "guildstash", name: "Guild Stash", icon: "🗄️", pageUrl: "/guildstash/ui" },
    { slug: "webui", name: "Item Browser", icon: "🔍", pageUrl: "/webui/db" }
  ];

  // Dev-only tools. Rendered in their own "Dev Utils" section pinned to the BOTTOM of the rail (it fills
  // upward from the bottom edge), separate from the mods above. Probed the same way as MODS.
  var DEV_TOOLS = [
    { slug: "webui", name: "Dev · Routes", icon: "🛠️", pageUrl: "/webui/dev/routes" },
    { slug: "webui", name: "Dev · Load Order", icon: "🔢", pageUrl: "/webui/dev/loadorder" }
  ];

  var RAIL_WIDTH = 210; // px, expanded
  var STRIP_WIDTH = 48; // px, collapsed to icons
  var COLLAPSE_KEY = "lwr-collapsed";

  // Does `pageUrl` answer? All mod listeners share this origin, so a relative GET either hits the
  // mod's real page (200) or, if the mod isn't installed, gets SPT's fallback (not ok). We only read
  // .ok, never the body.
  function probe(mod) {
    return fetch(mod.pageUrl, { method: "GET", cache: "no-store" })
      .then(function (res) { return res.ok ? mod : null; })
      .catch(function () { return null; });
  }

  // The non-Leaves mods that also ship a web UI, detected server-side (folder presence under
  // user/mods) and returned by LeavesUtils at /webui/rail/neighbors. We can't reliably probe these
  // from here ourselves - RaidReview in particular is a separate http server on another port, which an
  // https page can't fetch at all (mixed content) - so the server does the detection and we just render
  // whatever it reports. Each entry is { name, icon, url, external }; external ones open in a new tab.
  // Any failure (endpoint missing on an older LeavesUtils, bad JSON) resolves to an empty list.
  function fetchNeighbors() {
    return fetch("/webui/rail/neighbors", { method: "GET", cache: "no-store" })
      .then(function (res) { return res.ok ? res.json() : []; })
      .then(function (list) { return Array.isArray(list) ? list : []; })
      .catch(function () { return []; });
  }

  // Is this page the one a rail entry points at? Exact pathname match (trailing slash ignored), so two
  // entries that share a first path segment (e.g. /webui/db and /webui/dev/routes) don't both light up.
  function isActive(pageUrl) {
    var here = window.location.pathname.replace(/\/+$/, "");
    var target = (pageUrl || "").replace(/\/+$/, "");
    return here === target;
  }

  function injectStyles() {
    if (document.getElementById("lwr-styles")) {
      return;
    }
    var css =
      "#lwr-rail{position:fixed;top:0;right:0;height:100vh;width:" + RAIL_WIDTH + "px;" +
      "box-sizing:border-box;z-index:2147483000;display:flex;flex-direction:column;" +
      "background:#161616;border-left:1px solid #333;font-family:system-ui,-apple-system,Segoe UI,Roboto,sans-serif;" +
      "color:#ddd;transition:width .12s ease;overflow:hidden;}" +
      "#lwr-rail.lwr-collapsed{width:" + STRIP_WIDTH + "px;}" +
      "#lwr-head{display:flex;align-items:center;gap:8px;padding:10px 12px;border-bottom:1px solid #333;" +
      "font-size:12px;font-weight:600;letter-spacing:.04em;text-transform:uppercase;color:#8a8a8a;white-space:nowrap;}" +
      "#lwr-rail.lwr-collapsed #lwr-title{display:none;}" +
      "#lwr-toggle{margin-left:auto;cursor:pointer;background:none;border:none;color:#8a8a8a;font-size:16px;" +
      "line-height:1;padding:2px 4px;border-radius:4px;}" +
      "#lwr-toggle:hover{background:#2a2a2a;color:#ddd;}" +
      "#lwr-rail.lwr-collapsed #lwr-toggle{margin:0 auto;}" +
      "#lwr-list{list-style:none;margin:0;padding:6px;overflow-y:auto;flex:1;}" +
      "#lwr-devlist{list-style:none;margin:0;padding:6px;flex:0 0 auto;}" +
      ".lwr-item{display:flex;align-items:center;gap:10px;padding:9px 10px;margin:2px 0;border-radius:6px;" +
      "text-decoration:none;color:#ccc;font-size:14px;white-space:nowrap;cursor:pointer;}" +
      ".lwr-item:hover{background:#242424;color:#fff;}" +
      ".lwr-item.lwr-active{background:#2f3d2f;color:#cdeecd;font-weight:600;}" +
      ".lwr-icon{font-size:18px;line-height:1;flex:0 0 auto;width:20px;text-align:center;}" +
      "#lwr-rail.lwr-collapsed .lwr-label{display:none;}" +
      ".lwr-section{padding:10px 10px 4px;margin-top:6px;border-top:1px solid #2a2a2a;font-size:11px;" +
      "font-weight:600;letter-spacing:.04em;text-transform:uppercase;color:#6f6f6f;white-space:nowrap;}" +
      "#lwr-rail.lwr-collapsed .lwr-section{font-size:0;padding:6px 0;border-top:1px solid #2a2a2a;}" +
      "#lwr-empty{padding:12px;font-size:12px;color:#777;}";
    var style = document.createElement("style");
    style.id = "lwr-styles";
    style.textContent = css;
    document.head.appendChild(style);
  }

  function applyBodyOffset(width) {
    // Push the page over so nothing hides behind the rail. On ultrawide this is free space.
    document.body.style.marginRight = width + "px";
  }

  function build(installed, neighbors, devTools) {
    injectStyles();
    neighbors = neighbors || [];
    devTools = devTools || [];

    var collapsed = localStorage.getItem(COLLAPSE_KEY) === "1";

    var rail = document.createElement("nav");
    rail.id = "lwr-rail";
    if (collapsed) {
      rail.className = "lwr-collapsed";
    }

    var head = document.createElement("div");
    head.id = "lwr-head";
    var title = document.createElement("span");
    title.id = "lwr-title";
    title.textContent = "Leaves Mods";
    var toggle = document.createElement("button");
    toggle.id = "lwr-toggle";
    toggle.type = "button";
    toggle.title = "Collapse / expand";
    toggle.textContent = collapsed ? "«" : "»"; // « »
    head.appendChild(title);
    head.appendChild(toggle);
    rail.appendChild(head);

    var list = document.createElement("ul");
    list.id = "lwr-list";

    // Builds one row: an <a> with an icon + label. `active` highlights the current page; `external`
    // opens the link in a new tab (used for neighbor mods that live on their own port/server).
    function makeItem(href, name, iconText, active, external) {
      var item = document.createElement("a");
      item.className = "lwr-item";
      if (active) {
        item.className += " lwr-active";
      }
      item.href = href;
      item.title = name;
      if (external) {
        item.target = "_blank";
        item.rel = "noopener";
      }

      var icon = document.createElement("span");
      icon.className = "lwr-icon";
      icon.textContent = iconText;
      var label = document.createElement("span");
      label.className = "lwr-label";
      label.textContent = name;

      item.appendChild(icon);
      item.appendChild(label);
      return item;
    }

    if (installed.length === 0 && neighbors.length === 0 && devTools.length === 0) {
      var empty = document.createElement("li");
      empty.id = "lwr-empty";
      empty.textContent = "No mod UIs found.";
      list.appendChild(empty);
    } else {
      for (var i = 0; i < installed.length; i++) {
        var mod = installed[i];
        list.appendChild(makeItem(mod.pageUrl, mod.name, mod.icon, isActive(mod.pageUrl), false));
      }

      // Non-Leaves mods that also have a web UI, under their own "Other mods" heading. These are
      // outbound links (the neighbor page won't carry this rail back), and external ones open a new tab.
      if (neighbors.length > 0) {
        var section = document.createElement("li");
        section.className = "lwr-section";
        section.textContent = "Other mods";
        list.appendChild(section);

        for (var n = 0; n < neighbors.length; n++) {
          var nb = neighbors[n];
          list.appendChild(makeItem(nb.url, nb.name, nb.icon, false, nb.external === true));
        }
      }
    }
    rail.appendChild(list);

    // Dev tools live in their own list AFTER the scrollable mod list. Because the mod list is flex:1 it eats
    // the slack, so this block hugs the bottom of the rail and grows upward as tools are added.
    if (devTools.length > 0) {
      var devWrap = document.createElement("ul");
      devWrap.id = "lwr-devlist";

      var devSection = document.createElement("li");
      devSection.className = "lwr-section";
      devSection.textContent = "Dev Utils";
      devWrap.appendChild(devSection);

      for (var d = 0; d < devTools.length; d++) {
        var tool = devTools[d];
        devWrap.appendChild(makeItem(tool.pageUrl, tool.name, tool.icon, isActive(tool.pageUrl), false));
      }
      rail.appendChild(devWrap);
    }

    toggle.addEventListener("click", function () {
      var nowCollapsed = rail.classList.toggle("lwr-collapsed");
      localStorage.setItem(COLLAPSE_KEY, nowCollapsed ? "1" : "0");
      toggle.textContent = nowCollapsed ? "«" : "»";
      applyBodyOffset(nowCollapsed ? STRIP_WIDTH : RAIL_WIDTH);
    });

    document.body.appendChild(rail);
    applyBodyOffset(collapsed ? STRIP_WIDTH : RAIL_WIDTH);
  }

  // ---- Ctrl+F / Esc search shortcuts ---------------------------------------------------
  // Every mod page has one or more text search boxes. Ctrl+F (Cmd+F on Mac) focuses the page's
  // search box instead of opening the browser's native Find bar; Esc clears it and then steps
  // out. Tabbed pages (debug, reroll) keep several search inputs in the DOM but hide all but the
  // active tab's, so we only consider *visible* ones and naturally land on the right box. If a
  // page has no search box we DON'T preventDefault, so the browser's own Find still works there.

  function isVisible(el) {
    // offsetParent is null for anything display:none (including the hidden tabs' inputs), which
    // is exactly what we want to skip. None of our search boxes are position:fixed (that would
    // also null offsetParent), so this stays a reliable "is this actually on screen" check.
    return el.offsetParent !== null && el.getClientRects().length > 0;
  }

  function looksLikeSearch(el) {
    var hay = (el.id + " " + el.name + " " + el.placeholder + " " + el.className).toLowerCase();
    if (hay.indexOf("search") !== -1) { return true; }
    if (hay.indexOf("filter") !== -1) { return true; }
    if (hay.indexOf("find") !== -1) { return true; }
    if (el.id === "q") { return true; } // handbook's category filter is just id="q"
    return false;
  }

  // The page's current search box: the first visible text input that looks like a search field,
  // or failing that the first visible text input at all. null if the page has none.
  function findSearchBox() {
    var inputs = document.querySelectorAll("input");
    var firstText = null;
    for (var i = 0; i < inputs.length; i++) {
      var el = inputs[i];
      var type = (el.type || "text").toLowerCase();
      if (type !== "text" && type !== "search") {
        continue;
      }
      if (!isVisible(el)) {
        continue;
      }
      if (looksLikeSearch(el)) {
        return el;
      }
      if (firstText === null) {
        firstText = el;
      }
    }
    return firstText;
  }

  function onKeyDown(e) {
    var key = e.key ? e.key.toLowerCase() : "";

    // Ctrl/Cmd+F -> jump into the page's search box (select-all so typing replaces, like the
    // native Find field). Alt held = some other combo, leave it alone.
    if ((e.ctrlKey || e.metaKey) && !e.altKey && key === "f") {
      var box = findSearchBox();
      if (box === null) {
        return; // nothing to focus here - let the browser open its own Find bar
      }
      e.preventDefault();
      box.focus();
      box.select();
      return;
    }

    // Esc in a search box: first press clears it (which resets the page's filter to show
    // everything), a second press (now empty) blurs back out.
    if (key === "escape") {
      var active = document.activeElement;
      if (active && active.tagName === "INPUT" && looksLikeSearch(active)) {
        if (active.value !== "") {
          active.value = "";
          // Fire the same 'input' event a keystroke would, so the page's own filter re-runs.
          active.dispatchEvent(new Event("input", { bubbles: true }));
        } else {
          active.blur();
        }
      }
    }
  }

  function installSearchShortcuts() {
    document.addEventListener("keydown", onKeyDown);
  }

  function start() {
    installSearchShortcuts();
    var probes = [];
    for (var i = 0; i < MODS.length; i++) {
      probes.push(probe(MODS[i]));
    }
    var devProbes = [];
    for (var d = 0; d < DEV_TOOLS.length; d++) {
      devProbes.push(probe(DEV_TOOLS[d]));
    }
    // Probe the Leaves mods + dev tools and ask the server for installed neighbor mods, all at once.
    Promise.all([Promise.all(probes), fetchNeighbors(), Promise.all(devProbes)]).then(function (all) {
      var results = all[0];
      var neighbors = all[1];
      var devResults = all[2];
      var installed = [];
      for (var j = 0; j < results.length; j++) {
        if (results[j]) {
          installed.push(results[j]);
        }
      }
      var devTools = [];
      for (var k = 0; k < devResults.length; k++) {
        if (devResults[k]) {
          devTools.push(devResults[k]);
        }
      }
      // Nothing reachable at all (shouldn't happen - we're being served from one of them) - stay out
      // of the way rather than paint an empty rail.
      if (installed.length > 0 || neighbors.length > 0 || devTools.length > 0) {
        build(installed, neighbors, devTools);
      }
    });
  }

  if (document.readyState === "loading") {
    document.addEventListener("DOMContentLoaded", start);
  } else {
    start();
  }
})();
